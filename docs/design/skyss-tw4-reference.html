07 — Config file structure (skyss-tw4-shadcn.css)
/* ① Import Tailwind */ @import "tailwindcss"; /* ② All tokens live in @theme — no tailwind.config.js needed */ @theme { --font-sans: "Syne", sans-serif; --radius: 0rem; --color-amber-500: oklch(60% 0.180 56); /* ... all scales + semantic tokens ... */ --color-primary: var(--color-amber-500); --color-border: var(--color-navy-950); } /* ③ Base resets */ @layer base { * { border-color: var(--color-border); } body { background-color: var(--color-background); } } /* ④ P3 gamut override */ @supports (color: color(display-p3 0 0 0)) { @theme { --color-amber-500: color(display-p3 0.88 0.48 0.10); } } /* ⑤ Custom utilities */ @utility btn-brutal { border: 2px solid var(--color-border); box-shadow: 4px 4px 0 var(--color-border); &:hover { transform: translate(-2px, -2px); } &:active { transform: translate(2px, 2px); } }
